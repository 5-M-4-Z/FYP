function runHFSSScriptBatch(hfssExePath, scriptFilePath)
% runHFSSScriptBatch Launches HFSS to run a Python script, save S11, and exit.
% Compatible with older MATLAB versions (uses char instead of string).

    if nargin < 2
        error('Usage: runHFSSScriptBatch(hfssExePath, scriptFilePath)');
    end

    % Ensure the inputs are char
    if ~ischar(hfssExePath)
        hfssExePath = char(hfssExePath);
    end
    if ~ischar(scriptFilePath)
        scriptFilePath = char(scriptFilePath);
    end

    % Ensure paths are enclosed in double quotes
    if hfssExePath(1) ~= '"'
        hfssExePath = ['"', hfssExePath, '"'];
    end
    if scriptFilePath(1) ~= '"'
        scriptFilePath = ['"', scriptFilePath, '"'];
    end

    % Build command
    runScriptFlag = '-RunScript'; %RunScriptAndExit and RunScript another useful command
    cmd = [hfssExePath, ' ', runScriptFlag, ' ', scriptFilePath];

    fprintf('Running HFSS with command:\n%s\n', cmd);
    [status, cmdout] = system(cmd);

    if status ~= 0
        error('HFSS execution failed (status=%d). Output:\n%s', status, cmdout);
    end

    fprintf('HFSS script completed successfully.\n');
end
